# 60 fps: the community static 60 fps patch for 1.0.1 (MrComputerRevo's IPS). The IPS offsets count the 0x100-byte
# NSO header, so each lands at module offset -0x100 here. It halves the frame-time constants of the game's frame
# accumulator (1.0 -> 0.5 and 2.0 -> 1.0) and clears the argument that makes the frame call wait a second vsync.
# On by default; the menu's AC_FPS=30 (the ini's fps = 30) runs the original instructions, i.e. the game's shipping
# 30 fps.
set  fps60 FPS60 1
site fps60 main 3e39d0 320003e0 52800001   # orr w0, wzr, #1 -> mov w1, #0 (frame call's second argument)
site fps60 main 91dc94 1e2e1001 1e2c1001   # fmov s1, #1.0 -> #0.5 (frame-time clamp, low)
site fps60 main 91dca0 1e2e1001 1e2c1001   # fmov s1, #1.0 -> #0.5 (frame-time clamp, remainder)
site fps60 main 91dcac 32091be8 52a7e008   # 1.0f -> 0.5f (stored frame-time scale)
site fps60 main 91dccc 1e201001 1e2e1001   # fmov s1, #2.0 -> #1.0 (frame-time clamp, high)
site fps60 main 91dcd8 320203e8 32091be8   # 2.0f -> 1.0f (stored frame-time clamp)
